⚡ КОПИ-ПАСТ РЕШЕНИЕ:

  # В вашем main.py
  from zkill_panel import ZkillPanel
  
  class MainWindow:
      def __init__(self):
          self.zkill_panel = ZkillPanel()
          self.table.itemDoubleClicked.connect(self.on_click)
      
      def on_click(self, item):
          char_id = self.table.item(item.row(), 0).data(Qt.UserRole)
          char_name = self.table.item(item.row(), 0).text()
          self.zkill_panel.open_character_in_compact(char_id, char_name)

  # ГОТОВО! 🎉
